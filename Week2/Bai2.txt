Với các trường hợp giá trị int lớn đến mức tràn số, ta sẽ chuyển kiểu của sum thành long hoặc BigInteger.

int n = a.length;
for (int i = 0; i < n; i++) {
	for (int j = i+1; j < n; j++) {
                for (int k = j+1; k < n; k++) {	
			long sum = a[i] + a[j] + a[k];
			if (sum == 0) {
                        	StdOut.println(a[i] + " " + a[j] + " " + a[k]);
			}
		}
	}
}
